# Conteúdo do arquivo .env.local:

http://localhost:3001

# ║  [OPCIONAL] CONFIGURAR SERVIÇO DE IA                                     ║
# ╚═══════════════════════════════════════════════════════════════════════════╝

# ─────────────────────────────────────────────────────────────────────────────
# Opção A: Python + Flask + OpenAI
# ─────────────────────────────────────────────────────────────────────────────

# 1. Criar pasta para o serviço
mkdir ai-service
cd ai-service

# 2. Criar arquivo requirements.txt
echo flask==3.0.0 > requirements.txt
echo flask-cors==4.0.0 >> requirements.txt
echo openai==1.3.0 >> requirements.txt
echo python-dotenv==1.0.0 >> requirements.txt

# 3. Instalar dependências
pip install -r requirements.txt

# 4. Copiar código do exemplo (ver EXEMPLO_SERVICO_IA.md)
# Criar arquivo app.py com o código do exemplo

# 5. Executar serviço
python app.py

# ─────────────────────────────────────────────────────────────────────────────
# Opção B: Node.js + Express + OpenAI
# ─────────────────────────────────────────────────────────────────────────────

# 1. Criar pasta para o serviço
mkdir ai-service
cd ai-service

# 2. Inicializar projeto Node
npm init -y

# 3. Instalar dependências
npm install express cors openai dotenv

# 4. Copiar código do exemplo (ver EXEMPLO_SERVICO_IA.md)
# Criar arquivo server.js com o código do exemplo

# 5. Executar serviço
node server.js


# ╔═══════════════════════════════════════════════════════════════════════════╗
# ║  TESTAR SERVIÇO DE IA                                                     ║
# ╚═══════════════════════════════════════════════════════════════════════════╝

# Windows (PowerShell):
Invoke-RestMethod -Uri http://localhost:5000/validate -Method Post -ContentType "application/json" -Body '{"user_answer":"teste","correct_answer":"teste","question_context":"teste","question_id":"1"}'

# Linux/Mac:
curl -X POST http://localhost:5000/validate \
  -H "Content-Type: application/json" \
  -d '{"user_answer":"teste","correct_answer":"teste","question_context":"teste","question_id":"1"}'


# ╔═══════════════════════════════════════════════════════════════════════════╗

# ╔═══════════════════════════════════════════════════════════════════════════╗
# ║  COMANDOS DE DEBUG                                                        ║
# ╚═══════════════════════════════════════════════════════════════════════════╝

# Ver processos rodando nas portas
# Windows:
netstat -ano | findstr :3001
netstat -ano | findstr :5000

# Linux/Mac:
lsof -i :3001
lsof -i :5000

# Matar processo por porta
# Windows:
taskkill /PID [PID_NUMBER] /F

# Linux/Mac:
kill -9 [PID_NUMBER]

# Limpar cache do Next.js
rm -rf .next
npm run dev

# Reinstalar dependências
rm -rf node_modules
npm install


# ╔═══════════════════════════════════════════════════════════════════════════╗
# ║  COMANDOS DE BUILD E DEPLOY                                               ║
# ╚═══════════════════════════════════════════════════════════════════════════╝

# Build local
npm run build

# Rodar build localmente
npm start

# Deploy no Vercel
vercel --prod

# Ver logs do Vercel
vercel logs

# Cole: https://habilitadev-backend.onrender.com

vercel env add AI_VALIDATION_URL
# Cole a URL do seu serviço de IA hospedado

vercel env add AI_MODEL
# Cole: gpt-3.5-turbo

vercel env add AI_TEMPERATURE
# Cole: 0.3
